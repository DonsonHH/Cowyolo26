"""Write a source-backed current report; missing or failed cases remain incomplete."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import sys

ROOT = Path(__file__).parent
REPO = Path(r'C:\Users\Donson\Documents\Codex\exports\Cowyolo26-20261031\repo')
M = REPO/'maintenance/offline_counting'
revision = sys.argv[1]
load = lambda path: json.loads(path.read_text(encoding='utf-8-sig'))
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
write = lambda path, value: path.write_bytes(value.encode('utf-8'))
selected=[]
lines=[]
selection_file=M/f'REALTIME_SELECTION_{revision.upper()}.json'
selection=load(selection_file) if selection_file.exists() else None
local_videos=[]
for prefix, label, direction, role in [('aug14','Camera1 2023-08-14','DOWN','开发回归'),
                                     ('aug15','Camera1 2023-08-15','DOWN','快牛回归'),
                                     ('aug19','Camera2 2023-08-19','UP','安装配置回归'),
                                     ('aug20','Camera2 2023-08-20','UP','本轮调参后新日期')]:
    chosen=selection['cases'][prefix] if selection else {}
    tag=chosen.get('case',prefix+'_hd_s256_birth_'+revision)
    role=chosen.get('role',role)
    local_videos.append(tag+'_review.mp4')
    summary=M/'results'/(tag+'_summary.json')
    assessment=M/'results'/(tag+'_assessment.json')
    visual=M/'results'/(tag+'_visual_review.json')
    if not summary.exists() or not assessment.exists():
        selected.append({'case':tag,'status':'NOT_YET_TERMINALLY_ASSESSED'})
        lines.append(f'| {label} | {role} | 未完成 | — | — | — |')
        continue
    s,a=load(summary),load(assessment)
    count=a['actual_directional_counts'][direction]
    reference=a['user_directional_reference'][direction]
    review=load(visual) if visual.exists() else None
    reviewed=bool(review and review.get('pair_review_passed') and review['run_id']==s['run_id'] and
                  review['source_matches_summary'] and review['source_sha256']==s['video_sha256'] and
                  review['reviewed_event_ids']==[row['event_id'] for row in
                      [json.loads(line) for line in (M/'results'/(tag+'_events.jsonl')).read_text().splitlines() if line]] and
                  all(review['run_evidence_sha256'][name]==sha(M/'results'/(tag+'_'+name))
                      for name in ('summary.json','events.jsonl','frames.jsonl','status.json')))
    row={'case':tag,'role':role,'run_id':s['run_id'],'counts':s['counts'],'actual_direction':direction,
         'actual':count,'reference':reference,'source_duration_s':s['source_duration_s'],
         'stream_wall_s':s['stream_wall_elapsed_s'],'base_observations':s['base_model_forwards'],
         'total_model_forwards':s['model_forwards'],'birth_model_forwards':s.get('birth_refinement_forwards',0),
         'endpoint_model_forwards':s['endpoint_recovery_forwards'],
         'observations_per_wall_second':s['base_model_forwards']/s['stream_wall_elapsed_s'],
         'counts_and_service_pass':a['counts_and_service_pass'],'AI_pair_reviewed':reviewed,
         'assessment_path':str(assessment.relative_to(M)).replace('\\','/'),
         'summary_sha256':sha(summary),'assessment_sha256':sha(assessment),
         'event_p95_ms':a['event_publication_age'].get('p95_ms'),
         'event_max_ms':a['event_publication_age']['maximum_ms'],
         'observation_p95_upper_ms':a['observation_latency']['p95_upper_bound_ms']}
    # User required accurate wall-paced counting, not the agent-added 1s
    # event-publication limit. Preserve the original frozen-contract failure.
    row['posthoc_engineering_delivery_checks_pass']=bool(a['integrity_pass'] and a['count_reference_pass'] and
        all(value for key,value in a['clock_checks'].items() if key!='event_publication_deadline'))
    row['original_frozen_service_contract_pass']=a['service_clock_pass']
    selected.append(row)
    verdict='通过' if a['counts_and_service_pass'] else '失败'
    lines.append(f"| {label} | {role} | {direction} {count}/{reference} | {s['source_duration_s']:.1f} / {s['stream_wall_elapsed_s']:.2f}秒 | {row['observations_per_wall_second']:.2f}Hz | {verdict} |")
complete=len(selected)==4 and all(row.get('posthoc_engineering_delivery_checks_pass') and row.get('AI_pair_reviewed') for row in selected)
strict_service_complete=len(selected)==4 and all(row.get('original_frozen_service_contract_pass') for row in selected)
title=('现有四段视频数齐并跟上实时输入' if complete else '实时计数与图像核查尚未全部完成')
if complete and not strict_service_complete:
    title+='；原1秒事件上限仍有超时'
configuration_text=selection.get('configuration_description') if selection else None
configuration_text=configuration_text or (
    '四段使用同一算法预设：观测运动沿通道轴关联，按未完成目的端的阶段选择真实历史图像；弱候选须有近期轨迹或真实运动支持才触发当前图局部推理，成功补检后优先保留一张已到达图像。安装几何、方向及源范围随摄像头配置。Camera1复用原V14实测，逐项共享参数一致，原协议和评价保持原样，未按新协议重评；Camera2执行新的完整实机。'
    if selection else
    'Camera1使用观测运动并沿通道轴关联，按未完成目的端的阶段选择真实历史图像；弱候选须有近期轨迹或真实运动支持才触发当前图局部推理，成功补检后优先保留一张已到达图像。Camera2使用3Hz/Kalman安装配置，高分关联0.15、局部恢复最大年龄2秒。')
exposure_text=('**首次新日期冻结验证已经失败：Aug20 V14为9/12，361次真实模型调用，服务通过、数量未通过。** 后续关联诊断和统一配置选择使用了该结果，Aug20此后的测试全部属于开发回归，当前没有独立日期验证通过的证据。'
               if selection else
               'Aug20用于冻结后的新日期检验；首轮结果必须保留，若失败后继续调参则转为开发材料。')
erratum_text='V15冻结协议有一处继承文案，详见[SHA绑定勘误](REALTIME_BIRTH_V15_PROTOCOL_ERRATUM.json)：其中“untouched Aug20”是历史服务合同理由，V15的Aug20已暴露。原协议/冻结字节均保留，参数和评分未修改。' if (M/'REALTIME_BIRTH_V15_PROTOCOL_ERRATUM.json').exists() else ''
table='| 视频 | 用途 | 实测/人工参考 | 输入末PTS / 流水线时间 | 实际模型观察率 | 原冻结计数及1秒事件合同 |\n|---|---|---|---|---|---|\n'+'\n'.join(lines)
report=f'''# PYNQ-Z2：{title}

实验日期：2026-11-06。当前候选 `{revision}`。以下结果直接读取各运行的终态、摘要和外部协议评价；实时准确计数以现有材料的已知方向总数为验收范围。

用户要求是在板端实时完成计数，没有规定1秒事件上限。该上限由助手作为工程比较指标添加。本文将“数齐并跟上1倍输入”与“原冻结服务合同全部通过”分开：前者检查数量、完整性、采集、两帧队列、观察延迟及清空；后者仍保留原1秒事件判定，不能把超时改写为通过。全部逐事件AI图像核查完成才标记有界回放交付完成。

## 当前配置与依据

这是当前已有材料下选定的S256方案，不是所有摄像头的全局最优值。Camera2的2秒年龄参数同时用于端点恢复和起点补检的近期轨迹/弱运动支持；其起点ROI仍1.5倍，效果不能单独归因于端点。
